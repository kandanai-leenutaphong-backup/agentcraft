Which breaker alarms appeared for STN_A?

Write the query

let start = datetime(2024-04-15T10:00:00Z);
let end = datetime(2024-04-15T10:15:00Z);
WatcherAlarmsAll
| where station == "STN_A"
| where timestamp_aest between (start .. end)
| where alarm_label has "CB"
| project timestamp_aest, alarm_label, priority,
          client_composite_id, alarm_text
| order by timestamp_aest asc

Which equipment belongs to STN_A?

Write the query

g.V().hasLabel('Substation')
 .or(has('Substation.EMSName','STN_A'),
     has('IdentifiedObject.Name','STN_A'))
 .in('VoltageLevel.Substation').as('vl')
 .in('Equipment.EquipmentContainer')
 .dedup().limit(100)
 .project('voltage','label','name','mrid')
 .by(select('vl').values('IdentifiedObject.Name'))
 .by(label())
 .by(values('IdentifiedObject.Name'))
 .by(values('IdentifiedObject.CIMMRID'))
